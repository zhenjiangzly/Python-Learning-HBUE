### 第 3 题

```python
with open("poem.txt", "r", encoding="utf-8") as f:
    text = f.read()

words = text.split()
counts = {}
for w in words:
    counts[w] = counts.get(w, 0) + 1

print(f"总单词数：{len(words)}")

max_count = max(counts.values())
for w in words:                    # 按出现顺序找第一个次数最多的
    if counts[w] == max_count:
        print(f"出现最多的词：{w}，次数：{max_count}")
        break

seen = []
for w in words:
    if len(w) > 5 and w not in seen:
        seen.append(w)
print("长度大于5的单词：", seen)
```

### 第 4 题

```python
import jieba

text = input("请输入中文文本：")
words = jieba.lcut(text)
print(f"词数：{len(words)}")

counts = {}
for w in words:
    if len(w) > 1:
        counts[w] = counts.get(w, 0) + 1

top3 = sorted(counts.items(), key=lambda x: x[1], reverse=True)[:3]
for w, n in top3:
    print(f"{w} {n}")

with open("word_count.txt", "w", encoding="utf-8") as f:
    for w, n in counts.items():
        f.write(f"{w} {n}\n")
```
