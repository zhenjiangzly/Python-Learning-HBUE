### 第 1 题

```python
n = int(input("请输入n："))
total = 0
fact = 1
for i in range(1, n + 1):
    fact *= i          # 累积阶乘，避免重复计算
    total += fact

```

### 第 2 题

```python
s = input("请输入字符串：")
upper = lower = digit = 0
for ch in s:
    if ch.isupper():
        upper += 1
    elif ch.islower():
        lower += 1
    elif ch.isdigit():
        digit += 1
print(f"大写: {upper}, 小写: {lower}, 数字: {digit}")
```

### 第 3 题

```python
students = []
with open("scores.csv", "r", encoding="utf-8") as f:
    next(f)                     # 跳过表头
    for line in f:
        name, usual, final = line.strip().split(",")
        usual, final = float(usual), float(final)
        total = usual * 0.3 + final * 0.7
        students.append((name, total))

students.sort(key=lambda x: x[1], reverse=True)
fail = 0
for name, total in students:
    print(f"{name} {total:.1f}")
    if total < 60:
        fail += 1
print(f"不及格人数：{fail}")
```

### 第 4 题

```python
import turtle

turtle.speed(5)
colors = ["red", "orange", "green", "blue", "purple"]
side = 40

for i in range(5):
    turtle.pencolor(colors[i])
    turtle.penup()
    turtle.goto(-side / 2, -side / 2)   # 移动到左下角
    turtle.pendown()
    for _ in range(4):
        turtle.forward(side)
        turtle.left(90)
    side += 40

turtle.done()
```
