### 第 2 题

```python
s = input("请输入字符串：")
counts = {}
for ch in s:
    counts[ch] = counts.get(ch, 0) + 1

max_count = max(counts.values())
for ch in s:                     # 按原顺序找第一个
    if counts[ch] == max_count:
        print(ch, max_count)
        break
```

### 第 3 题

```python
scores = []
with open("scores.txt", "r", encoding="utf-8") as f:
    for line in f:
        s = line.strip()
        if not s:
            continue
        try:
            score = int(s)
        except ValueError:
            continue
        if 0 <= score <= 100:
            scores.append(score)

if scores:
    avg = sum(scores) / len(scores)
    print(f"平均分：{avg:.1f}")
    print(f"最高分：{max(scores)}")
    print(f"最低分：{min(scores)}")

    excellent = sum(1 for s in scores if s >= 90)
    pass_count = sum(1 for s in scores if 60 <= s < 90)
    fail_count = sum(1 for s in scores if s < 60)

    with open("statistics.txt", "w", encoding="utf-8") as f:
        f.write(f"优秀 {excellent}\n")
        f.write(f"及格 {pass_count}\n")
        f.write(f"不及格 {fail_count}\n")
```

### 第 4 题

```python
import turtle

turtle.speed(10)

# 中心圆
turtle.pencolor("orange")
turtle.fillcolor("yellow")
turtle.begin_fill()
turtle.circle(30)
turtle.end_fill()

# 12 片花瓣
for i in range(12):
    if i % 2 == 0:
        turtle.pencolor("red")
    else:
        turtle.pencolor("orange")
    turtle.circle(60, 60)     # 画 60° 弧
    turtle.left(30)           # 每片旋转 30°

turtle.done()
```
